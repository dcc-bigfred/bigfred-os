# Raspberry Pi 5 — boot firmware (BigFred hub)
# Pi 5 boots from EEPROM; start*.elf / fixup*.dat are not used.

kernel=Image

# Hide firmware rainbow splash. Graphical BigFred logo (fullscreen_logo + logo.cpio)
# is disabled so boot shows a plain text console (microinit / kernel messages).
disable_splash=1
# initramfs logo.cpio followkernel

# Pi 5 Rev 1.1 (BCM2712 D0) — required. Using the C0 DTB on D0 silicon
# panics in bcm2712_pull_config_set during UART/pinctrl probe (SError).
# For older Rev 1.0 (C0) boards, change to: bcm2712-rpi-5-b.dtb
device_tree=bcm2712d0-rpi-5-b.dtb

# Pi 5: enable PCIe Gen3 for NVMe HAT+
dtparam=pciex1_gen=3

# BCM54213PE AutogrEEEn + macb EEE default-on on 6.18 = link flaps.
# dtparam=eee=off leaves eee-broken-1000t/100tx on the PHY (Pi 5 / CM5).
dtparam=eee=off

# Active cooler — kernel pwm-fan driver + thermal framework (hub spec §8.8).
# The kernel auto-controls the PWM via cooling_device0 using the
# thresholds/speeds below (millidegrees Celsius, PWM 0-255).
# Requires CONFIG_SENSORS_PWM_FAN=y (hub-kernel linux-hub.fragment).
# Levels: <35 OFF | 35–60 LOW (75) | 60–70 MED (150) | >70 HIGH (255).
dtparam=cooling_fan=on
dtparam=fan_temp0=35000,fan_temp0_hyst=5000,fan_temp0_speed=75
dtparam=fan_temp1=60000,fan_temp1_hyst=5000,fan_temp1_speed=150
dtparam=fan_temp2=70000,fan_temp2_hyst=5000,fan_temp2_speed=255
dtparam=fan_temp3=70000,fan_temp3_hyst=5000,fan_temp3_speed=255

# UART: serial console on Pi 5 debug UART (ttyAMA10)
enable_uart=1

# 64-bit only
arm_64bit=1
